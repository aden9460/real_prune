### 8.2 验证 A：V 矩阵主成分分析

**目的**：检查 V 矩阵是否有频域分离结构

```python
def analyze_v_matrix_structure(model, layer_idx):
    """
    分析 V projection 的主成分结构
    """
    # 提取 V projection weight
    qkv_weight = model.blocks[layer_idx].attn.mat_qkv.weight
    hidden_dim = qkv_weight.shape[1]

    # V 对应的是第三个 1/3
    V_weight = qkv_weight[2*hidden_dim:3*hidden_dim, :]  # [hidden_dim, hidden_dim]

    # SVD 分解
    U, S, Vt = torch.svd(V_weight)

    # 分析特征值分布
    variance_explained = S**2 / (S**2).sum()
    cumsum_variance = torch.cumsum(variance_explained, dim=0)

    # 绘图
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))

    # Plot 1: 特征值分布
    ax = axes[0]
    ax.plot(S.cpu().numpy(), 'o-')
    ax.set_xlabel('Component')
    ax.set_ylabel('Singular Value')
    ax.set_title(f'Layer {layer_idx}: V Matrix Singular Values')
    ax.set_yscale('log')
    ax.grid(True, alpha=0.3)

    # Plot 2: 累积方差解释率
    ax = axes[1]
    ax.plot(cumsum_variance.cpu().numpy())
    ax.axhline(y=0.9, color='r', linestyle='--', label='90% variance')
    ax.set_xlabel('Number of Components')
    ax.set_ylabel('Cumulative Variance Explained')
    ax.set_title(f'Layer {layer_idx}: Cumulative Variance')
    ax.legend()
    ax.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig(f'v_matrix_analysis_layer{layer_idx}.png', dpi=200)
    plt.close()

    # 统计
    n_components_90 = torch.sum(cumsum_variance < 0.9).item() + 1
    effective_rank = (S.sum()**2 / (S**2).sum()).item()

    print(f"Layer {layer_idx}:")
    print(f"  Components for 90% variance: {n_components_90}/{len(S)}")
    print(f"  Effective rank: {effective_rank:.1f}")
    print(f"  Rank ratio: {effective_rank/len(S):.2%}")

    # 判断是否有明显的低秩结构
    if n_components_90 < len(S) * 0.5:
        print(f"  ✓ V has low-rank structure (may support block-diagonal)")
    else:
        print(f"  ✗ V is full-rank (less support for block-diagonal)")

    return S, U, Vt

# 运行
for layer_idx in [0, 5, 7, 11]:
    analyze_v_matrix_structure(model, layer_idx)
```

**解读**：

- **有频域分离**：前 k 个主成分解释 > 90% 方差，且 k 较小
- **无频域分离**：接近满秩，所有主成分都重要

### 8.3 验证 B：Head Output Correlation Matrix

**目的**：直接测量不同 heads 输出的相关性

```python
def compute_head_output_correlation(model, calibration_data, layer_idx):
    """
    计算 head outputs 的相关性矩阵

    这是 Hessian 的归一化版本
    """
    # 1. 收集所有 heads 的输出
    all_head_outputs = [[] for _ in range(num_heads)]

    for sample in tqdm(calibration_data, desc=f"Layer {layer_idx}"):
        # Forward pass 到目标层
        x = model.forward_to_layer(sample, layer_idx)

        # 提取 attention layer 的输出（在 O projection 之前）
        attn_layer = model.blocks[layer_idx].attn

        # 手动计算 attention
        B, L, C = x.shape
        qkv = F.linear(x, attn_layer.mat_qkv.weight,
                      torch.cat([attn_layer.q_bias, attn_layer.zero_k_bias, attn_layer.v_bias]))
        qkv = qkv.view(B, L, 3, attn_layer.num_heads, 64)
        q, k, v = qkv.permute(2, 0, 3, 1, 4).unbind(0)  # [B, H, L, 64]

        # L2 normalize
        if attn_layer.attn_l2_norm:
            scale_mul = attn_layer.scale_mul_1H11.clamp_max(attn_layer.max_scale_mul).exp()
            q = F.normalize(q, dim=-1).mul(scale_mul)
            k = F.normalize(k, dim=-1)

        # Attention
        attn_weights = torch.softmax(q @ k.transpose(-2, -1) * attn_layer.scale, dim=-1)
        head_outputs = attn_weights @ v  # [B, H, L, 64]

        # 收集每个 head 的输出
        for h in range(num_heads):
            all_head_outputs[h].append(head_outputs[:, h].reshape(-1, 64))  # [B*L, 64]

    # 2. 拼接
    for h in range(num_heads):
        all_head_outputs[h] = torch.cat(all_head_outputs[h], dim=0)  # [N*L, 64]

    # 3. 计算 pairwise correlation
    correlation_matrix = torch.zeros(num_heads, num_heads)

    for i in range(num_heads):
        for j in range(num_heads):
            # 计算协方差矩阵的 Frobenius norm
            out_i = all_head_outputs[i]  # [N, 64]
            out_j = all_head_outputs[j]  # [N, 64]

            cov = (out_i.T @ out_j) / len(out_i)  # [64, 64]
            correlation_matrix[i, j] = torch.norm(cov, 'fro')

    # 4. 归一化为相关系数
    for i in range(num_heads):
        for j in range(num_heads):
            correlation_matrix[i, j] /= torch.sqrt(
                correlation_matrix[i, i] * correlation_matrix[j, j]
            )

    # 5. 可视化
    plt.figure(figsize=(10, 8))
    sns.heatmap(correlation_matrix.cpu().numpy(),
                cmap='RdBu_r', center=0, vmin=-1, vmax=1,
                annot=True, fmt='.2f', cbar_kws={'label': 'Correlation'})
    plt.title(f'Layer {layer_idx}: Head Output Correlation Matrix')
    plt.xlabel('Head Index')
    plt.ylabel('Head Index')
    plt.tight_layout()
    plt.savefig(f'head_correlation_layer{layer_idx}.png', dpi=200)
    plt.close()

    return correlation_matrix
```

### 8.4 验证 C：块对角性定量测量（BDR）

**目的**：测量 Hessian 的块对角程度

```python
def measure_block_diagonality(correlation_matrix, head_groups, layer_idx):
    """
    计算 Block Diagonality Ratio (BDR)

    BDR = ||H_cross|| / sqrt(||H_00|| · ||H_11||)

    BDR < 0.3: 强块对角（理想）
    BDR < 0.5: 弱块对角（可接受）
    BDR > 0.5: 不块对角（风险高）
    """
    num_heads = correlation_matrix.shape[0]

    # 分组
    group_0 = [h for h in range(num_heads)
               if head_groups.get(f"layer_{layer_idx}_head_{h}", 0) == 0]
    group_1 = [h for h in range(num_heads)
               if head_groups.get(f"layer_{layer_idx}_head_{h}", 0) == 1]

    # 提取块
    H_00 = correlation_matrix[group_0, :][:, group_0]
    H_11 = correlation_matrix[group_1, :][:, group_1]
    H_01 = correlation_matrix[group_0, :][:, group_1]
    H_10 = correlation_matrix[group_1, :][:, group_0]

    # 计算 Frobenius norm
    norm_00 = torch.norm(H_00, 'fro')
    norm_11 = torch.norm(H_11, 'fro')
    norm_01 = torch.norm(H_01, 'fro')
    norm_10 = torch.norm(H_10, 'fro')
    norm_cross = (norm_01 + norm_10) / 2

    # Block Diagonality Ratio
    BDR = norm_cross / torch.sqrt(norm_00 * norm_11)

    # 组内平均相关性 vs 组间平均相关性
    within_corr_0 = H_00.mean()
    within_corr_1 = H_11.mean()
    cross_corr = H_01.mean()

    print(f"\nLayer {layer_idx} Block Diagonality Analysis:")
    print(f"  Group sizes: {len(group_0)} vs {len(group_1)}")
    print(f"  ||H_00||_F = {norm_00:.2f}")
    print(f"  ||H_11||_F = {norm_11:.2f}")
    print(f"  ||H_01||_F = {norm_01:.2f}")
    print(f"  ||H_10||_F = {norm_10:.2f}")
    print(f"  BDR = {BDR:.3f}")
    print(f"  Within-group avg corr: {(within_corr_0 + within_corr_1)/2:.3f}")
    print(f"  Cross-group avg corr: {cross_corr:.3f}")

    if BDR < 0.3:
        status = "✓ STRONG block diagonal (BDR < 0.3)"
        verdict = "proceed"
    elif BDR < 0.5:
        status = "⚠ WEAK block diagonal (0.3 < BDR < 0.5)"
        verdict = "caution"
    else:
        status = "✗ NOT block diagonal (BDR > 0.5)"
        verdict = "abort"

    print(f"  {status}")
    print(f"  Recommendation: {verdict.upper()}")

    return {
        'BDR': BDR.item(),
        'norm_00': norm_00.item(),
        'norm_11': norm_11.item(),
        'norm_cross': norm_cross.item(),
        'within_corr': (within_corr_0.item() + within_corr_1.item()) / 2,
        'cross_corr': cross_corr.item(),
        'verdict': verdict
    }
```

### 8.5 验证 D：重建误差对比（最关键！）

**目的**：直接测试组内 vs 全局补偿的效果

```python
def verify_reconstruction_error(model, calibration_data, head_groups,
                               layer_idx, sparsity=0.2):
    """
    最关键的验证：比较重建误差

    如果 MSE_grouped >= MSE_global，说明组内补偿不优于全局
    → 应该放弃完整实验
    """
    import copy

    device = next(model.parameters()).device

    # 1. 收集原始输出（ground truth）
    print(f"\n{'='*60}")
    print(f"Verification D: Layer {layer_idx} Reconstruction Error")
    print(f"{'='*60}\n")

    print("Step 1: Collecting original outputs...")
    original_outputs = []
    for batch in tqdm(calibration_data, desc="Original"):
        with torch.no_grad():
            out = model.blocks[layer_idx](batch.to(device))
            original_outputs.append(out.cpu())
    original_outputs = torch.cat(original_outputs, dim=0)

    # 2. 收集 attention outputs（用于 SlimGPT）
    print("\nStep 2: Collecting attention outputs...")
    attention_outputs = []
    for batch in tqdm(calibration_data, desc="Attention"):
        with torch.no_grad():
            # Forward 到 O projection 之前
            x = model.forward_to_layer(batch.to(device), layer_idx)
            attn_layer = model.blocks[layer_idx].attn

            # 计算 attention output（O projection 的输入）
            B, L, C = x.shape
            qkv = F.linear(x, attn_layer.mat_qkv.weight,
                          torch.cat([attn_layer.q_bias, attn_layer.zero_k_bias, attn_layer.v_bias]))
            qkv = qkv.view(B, L, 3, attn_layer.num_heads, 64)
            q, k, v = qkv.permute(2, 0, 3, 1, 4).unbind(0)

            if attn_layer.attn_l2_norm:
                scale_mul = attn_layer.scale_mul_1H11.clamp_max(attn_layer.max_scale_mul).exp()
                q = F.normalize(q, dim=-1).mul(scale_mul)
                k = F.normalize(k, dim=-1)

            attn_weights = torch.softmax(q @ k.transpose(-2, -1) * attn_layer.scale, dim=-1)
            attn_out = attn_weights @ v  # [B, H, L, 64]
            attn_out = attn_out.transpose(1, 2).reshape(B, L, -1)  # [B, L, H*64]

            attention_outputs.append(attn_out.cpu())

    attention_outputs = torch.cat(attention_outputs, dim=0)

    # 3. 全局补偿剪枝
    print("\nStep 3: Testing GLOBAL compensation...")
    model_global = copy.deepcopy(model)

    # 创建 SlimGPT
    from slim_utils.slimgpt import SlimGPT
    O_weight_global = model_global.blocks[layer_idx].attn.proj.weight.data.clone()
    slimgpt_global = SlimGPT(O_weight_global)

    # 添加激活
    slimgpt_global.add_batch(attention_outputs.reshape(-1, attention_outputs.shape[-1]))

    # 全局剪枝
    pruned_heads_global = slimgpt_global.struct_prune(sparsity=sparsity, headsize=64)

    # 应用到模型
    model_global.blocks[layer_idx].attn.proj.weight.data = slimgpt_global.W

    # 测试输出
    outputs_global = []
    for batch in tqdm(calibration_data, desc="Global output"):
        with torch.no_grad():
            out = model_global.blocks[layer_idx](batch.to(device))
            outputs_global.append(out.cpu())
    outputs_global = torch.cat(outputs_global, dim=0)

    mse_global = F.mse_loss(outputs_global, original_outputs)

    # 4. 组内补偿剪枝
    print("\nStep 4: Testing GROUPED compensation...")
    model_grouped = copy.deepcopy(model)

    O_weight_grouped = model_grouped.blocks[layer_idx].attn.proj.weight.data.clone()
    slimgpt_grouped = SlimGPT(O_weight_grouped)
    slimgpt_grouped.add_batch(attention_outputs.reshape(-1, attention_outputs.shape[-1]))

    # 组内剪枝
    pruned_heads_grouped = slimgpt_grouped.struct_prune_grouped(
        sparsity=sparsity,
        head_groups=head_groups,
        layer_idx=layer_idx,
        headsize=64
    )

    model_grouped.blocks[layer_idx].attn.proj.weight.data = slimgpt_grouped.W

    outputs_grouped = []
    for batch in tqdm(calibration_data, desc="Grouped output"):
        with torch.no_grad():
            out = model_grouped.blocks[layer_idx](batch.to(device))
            outputs_grouped.append(out.cpu())
    outputs_grouped = torch.cat(outputs_grouped, dim=0)

    mse_grouped = F.mse_loss(outputs_grouped, original_outputs)

    # 5. 对比分析
    print(f"\n{'='*60}")
    print("RESULTS:")
    print(f"{'='*60}")
    print(f"MSE (Global):  {mse_global:.6f}")
    print(f"MSE (Grouped): {mse_grouped:.6f}")
    print(f"Difference:    {(mse_grouped - mse_global):.6f}")
    print(f"Relative:      {(mse_grouped / mse_global - 1) * 100:+.1f}%")

    improvement_threshold = 0.95  # Grouped 需要比 Global 好 5% 才算通过

    if mse_grouped < mse_global * improvement_threshold:
        print(f"\n✅ PASS: Grouped compensation is significantly better!")
        print(f"   → Proceed with full experiment")
        verdict = "proceed"
    elif mse_grouped < mse_global * 1.05:
        print(f"\n⚠️  MARGINAL: Grouped is about the same as global")
        print(f"   → Consider hybrid approach or abort")
        verdict = "marginal"
    else:
        print(f"\n❌ FAIL: Grouped compensation is WORSE than global!")
        print(f"   → ABORT full experiment, try alternative directions")
        verdict = "abort"

    print(f"{'='*60}\n")

    return {
        'mse_global': mse_global.item(),
        'mse_grouped': mse_grouped.item(),
        'relative_diff': (mse_grouped / mse_global - 1).item(),
        'verdict': verdict
    }
```

### 11.2 方向 2：Attention Pattern Redundancy Analysis（推荐度：★★★★☆）

**核心想法**：
- 直接测量 attention patterns 的冗余度
- 优先剪掉 pattern 相似的 heads

**方法**：

```python
def compute_attention_pattern_similarity(model, calibration_data, layer_idx):
    """
    构建 attention pattern 相似度矩阵
    """
    # 提取所有 heads 的 attention maps
    all_attn = extract_all_attention(model, layer_idx, calibration_data)
    # all_attn: [num_heads, N, L, L]

    # 计算 pairwise cosine similarity
    similarity_matrix = torch.zeros(num_heads, num_heads)

    for i in range(num_heads):
        for j in range(i+1, num_heads):
            attn_i = all_attn[i].reshape(-1)
            attn_j = all_attn[j].reshape(-1)

            sim = F.cosine_similarity(attn_i, attn_j, dim=0)
            similarity_matrix[i, j] = sim
            similarity_matrix[j, i] = sim

    # 冗余度 = 与其他 heads 的平均相似度
    redundancy_score = similarity_matrix.mean(dim=1)

    return redundancy_score

def prune_by_redundancy(model, sparsity):
    """
    结合 SlimGPT error 和 attention redundancy
    """
    for layer_idx in range(num_layers):
        slimgpt_error = compute_slimgpt_error(layer_idx)
        redundancy = compute_attention_pattern_similarity(layer_idx)

        # 综合评分：错误低 + 冗余度高 → 优先剪
        score = -slimgpt_error * 0.7 + redundancy * 0.3

        pruned_heads = select_top_k(score, k=int(num_heads * sparsity))
        slimgpt.struct_prune_with_heads(pruned_heads)
```

**优势**：
- ✅ 直接测量 pattern 冗余（符合直觉）
- ✅ 可解释性强
- ✅ 结合两种信息（SlimGPT + Attention）

**预期效果**：改进 3-6%

### 11.3 方向 3：Layer-wise Adaptive Pruning Strategy（推荐度：★★★★☆）

**核心想法**：
- 不是所有层用同样的剪枝率
- 基于 attention 的统计特性决定每层的剪枝率

**方法**：

```python
def analyze_layer_attention_properties(model, calibration_data):
    """
    分析每层 attention 的特性
    """
    layer_properties = {}

    for layer_idx in range(num_layers):
        attn = extract_attention(model, layer_idx, calibration_data)

        # 特性 1: Head diversity（头之间的多样性）
        diversity = compute_head_diversity(attn)
        # 高 → 难剪，低 → 易剪

        # 特性 2: Average entropy
        avg_entropy = compute_average_entropy(attn)
        # 高 → 关注分散（可能冗余）→ 易剪

        # 特性 3: Attention locality
        locality = compute_attention_locality(attn)
        # 高 → 类似 CNN（冗余）→ 易剪

        layer_properties[layer_idx] = {
            'diversity': diversity,
            'entropy': avg_entropy,
            'locality': locality
        }

    return layer_properties

def adaptive_sparsity_allocation(layer_properties, global_sparsity=0.2):
    """
    根据层特性分配剪枝率
    """
    sparsity_dict = {}

    for layer_idx, props in layer_properties.items():
        # 可剪程度
        pruneability = (
            (1 - props['diversity']) * 0.4 +
            props['entropy'] * 0.3 +
            props['locality'] * 0.3
        )

        # 分配剪枝率（可剪层剪更多）
        sparsity_dict[layer_idx] = base_sparsity * (0.5 + pruneability)

    # 归一化，保证全局剪枝率
    normalize_to_global(sparsity_dict, global_sparsity)

    return sparsity_dict
```

**示例结果**：

```
Layer 0:  diversity=0.3 → sparsity=0.30（易剪）
Layer 5:  diversity=0.7 → sparsity=0.10（难剪）
Layer 11: diversity=0.5 → sparsity=0.20（中等）
```

**优势**：
- ✅ 符合"不同层有不同冗余度"的直觉
- ✅ 数据驱动，自动发现模式
- ✅ 可以保护关键层

**预期效果**：改进 4-8%（有一定风险但潜力大）

### 11.4 方向 4：Attention-Weighted Soft Compensation（推荐度：★★★☆☆）

**核心想法**：
- 不是硬分组（0/1），而是软权重
- 基于 attention similarity 决定补偿权重

**方法**：

```python
def attention_weighted_compensation(pruned_head, remaining_heads):
    """
    基于 attention similarity 加权补偿
    """
    attn_pruned = get_attention_pattern(pruned_head)

    weights = []
    for remain_head in remaining_heads:
        attn_remain = get_attention_pattern(remain_head)

        # Cosine similarity
        sim = F.cosine_similarity(attn_pruned.reshape(-1),
                                 attn_remain.reshape(-1), dim=0)

        # 转换为权重
        weight = torch.exp(sim / temperature)
        weights.append(weight)

    weights = F.normalize(torch.tensor(weights), p=1, dim=0)

    # 加权补偿
    for i, remain_head in enumerate(remaining_heads):
        compensation = compute_slimgpt_compensation(pruned_head, remain_head)
        apply_compensation(remain_head, weights[i] * compensation)
```

**优势**：
- ✅ 避免硬分组的错误
- ✅ 优雅地退化到全局补偿

**预期效果**：改进 2-4%
